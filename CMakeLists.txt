# Code Modification - Build System: Establish the versioned WiseEngine build baseline.
# Scope: engine targets. Performance: no runtime cost. Toggle: WISEENGINE_* feature options.
cmake_minimum_required(VERSION 3.31)
project(WiseEngine VERSION 0.0.3 DESCRIPTION "Diligent and Luau outdoor game engine" LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "WiseEngine supports Windows x64 with MSVC only.")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release|RelWithDebInfo)$")
    message(FATAL_ERROR "Choose Debug, Release or RelWithDebInfo using a supplied preset.")
endif()
option(WISEENGINE_BUILD_TESTS "Build milestone and SDK integration tests" OFF)
option(WISEENGINE_ENABLE_PROFILING "Enable Tracy (CPU, memory and network overhead)" OFF)
option(WISEENGINE_ENABLE_VULKAN "Build Vulkan SDK support alongside DX12" ON)
option(WISEENGINE_BUILD_ASSET_TOOLS "Expose explicitly built asset and language tools" ON)
option(WISEENGINE_JOLT_DOUBLE_PRECISION "Double physics positions (CPU/memory cost)" ON)
option(WISEENGINE_ENABLE_FUTURE_SDKS "Expose optional ozz and Recast targets" OFF)
add_library(WiseEngineBuildOptions INTERFACE)
target_compile_features(WiseEngineBuildOptions INTERFACE cxx_std_20)
target_compile_definitions(WiseEngineBuildOptions INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(WiseEngineBuildOptions INTERFACE
    /W4 /WX /permissive- /FS /Zc:preprocessor /Zc:__cplusplus /diagnostics:column /utf-8)
function(wiseengine_target name)
    target_link_libraries(${name} PRIVATE WiseEngineBuildOptions)
endfunction()
add_subdirectory(thirdparty)
include(cmake/ModuleBoundaries.cmake)
add_subdirectory(src/Math)
add_subdirectory(src/Core)
foreach(module Renderer Physics Audio Input Network AssetManager)
    add_subdirectory(src/${module})
endforeach()
add_subdirectory(src/App)
wiseengine_check_module_boundaries()
if(WISEENGINE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
message(STATUS "WiseEngine ${PROJECT_VERSION}: ${CMAKE_BUILD_TYPE}, C++20, MSVC x64")
# End of Code Modification - Build System
